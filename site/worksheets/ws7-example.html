<?php
// 7.2 ฟังก์ชันแบบระบุชนิดข้อมูล และ match() ของ PHP 8
declare(strict_types=1);   // ต้องอยู่บรรทัดแรกของไฟล์ ให้ PHP ตรวจชนิดข้อมูลแบบเข้มงวด

/** คืนอัตราส่วนลดตามระดับสมาชิก */
function discountRate(string $level): float
{
    return match ($level) {
        'gold'   => 0.15,
        'silver' => 0.10,
        'member' => 0.05,
        default  => 0.0,     // ผู้ใช้ทั่วไปหรือระดับที่ไม่รู้จัก
    };
}

/** คืนราคาหลังหักส่วนลด ปัดเป็นทศนิยม 2 ตำแหน่ง */
function finalPrice(float $price, string $level): float
{
    if ($price < 0) {
        throw new InvalidArgumentException('ราคาต้องไม่ติดลบ');
    }
    return round($price * (1 - discountRate($level)), 2);
}

7.1

ฟอร์มคำนวณเกรดด้วย $_GET และ $_POST

ส่วนที่ 2 · สัปดาห์ 13

ใช้เกณฑ์เกรดตามตัวอย่าง If Else-If ในหัวข้อ 7.7 ของบทที่ 7 (A ≥ 80, B ≥ 70, C ≥ 60, D ≥ 50) แล้วย้ายไปไว้ในฟังก์ชัน หน้าเดียวกันรับค่าทั้งสองแบบ: เลือกวิชาจากลิงก์ด้วย $_GET และส่งชื่อกับคะแนนด้วย $_POST

<?php
// 7.1 รับค่าด้วย $_GET และ $_POST แล้วคำนวณเกรดด้วยฟังก์ชัน
declare(strict_types=1);

/** แปลงคะแนน 0–100 เป็นเกรด (เกณฑ์ตามตัวอย่างในบทที่ 7) */
function gradeFromScore(float $score): string
{
    if ($score >= 80) {
        return 'A';
    } elseif ($score >= 70) {
        return 'B';
    } elseif ($score >= 60) {
        return 'C';
    } elseif ($score >= 50) {
        return 'D';
    }
    return 'F';
}

// $_GET: เลือกวิชาจากลิงก์ ค่าอยู่ใน URL เช่น grade.php?subject=database
$subjects = ['web' => 'การเขียนโปรแกรมบนเว็บ', 'database' => 'ระบบฐานข้อมูล'];
$subjectKey = $_GET['subject'] ?? 'web';
if (!array_key_exists($subjectKey, $subjects)) {
    $subjectKey = 'web';                    // รับเฉพาะค่าที่อยู่ในรายการ
}

$name = '';
$scoreInput = '';
$result = null;
$error = '';

// $_POST: รับข้อมูลจากฟอร์ม ค่าไม่แสดงใน URL
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $name = trim($_POST['name'] ?? '');
    $scoreInput = trim($_POST['score'] ?? '');

    if ($name === '') {
        $error = 'กรุณากรอกชื่อ';
    } elseif (!is_numeric($scoreInput) || $scoreInput < 0 || $scoreInput > 100) {
        $error = 'คะแนนต้องเป็นตัวเลข 0–100';
    } else {
        $result = gradeFromScore((float) $scoreInput);
    }
}

$title = 'คำนวณเกรด';
require_once 'includes/header.php';
?>

</nav>

  <label for="score">คะแนน (0–100)</label>
  <input type="number" id="score" name="score" min="0" max="100" step="0.5" value="<?= htmlspecialchars($scoreInput) ?>" required>

โค้ด
<?php
// ฟังก์ชันจัดการผู้ใช้ เก็บในไฟล์ JSON ชั่วคราว (บทที่ 8 จะเปลี่ยนไปใช้ฐานข้อมูล MySQL)
declare(strict_types=1);

const USERS_FILE = __DIR__ . '/../data/users.json';

/** อ่านผู้ใช้ทั้งหมด คืนอาร์เรย์ [username => passwordHash] */
function loadUsers(): array
{
    if (!file_exists(USERS_FILE)) {
        return [];
    }
    $data = json_decode((string) file_get_contents(USERS_FILE), true);
    return is_array($data) ? $data : [];
}

/** สมัครสมาชิก: คืนข้อความผิดพลาด หรือสตริงว่างถ้าสำเร็จ */
function registerUser(string $username, string $password, string $confirm): string
{
    if (!preg_match('/^[a-z0-9_]{4,20}$/', $username)) {
        return 'ชื่อผู้ใช้ต้องเป็น a–z, 0–9 หรือ _ ยาว 4–20 ตัว';
    }
    if (strlen($password) < 8) {
        return 'รหัสผ่านต้องยาวอย่างน้อย 8 ตัว';
    }
    if ($password !== $confirm) {
        return 'รหัสผ่านทั้งสองช่องไม่ตรงกัน';
    }
    $users = loadUsers();
    if (isset($users[$username])) {
        return 'ชื่อผู้ใช้นี้ถูกใช้แล้ว';
    }
    // เก็บเฉพาะค่าแฮช ไม่เก็บรหัสผ่านจริง
    $users[$username] = password_hash($password, PASSWORD_DEFAULT);
    file_put_contents(USERS_FILE, json_encode($users, JSON_PRETTY_PRINT), LOCK_EX);
    return '';
}

/** ตรวจชื่อผู้ใช้และรหัสผ่าน คืน true ถ้าถูกต้อง */
function verifyLogin(string $username, string $password): bool
{
    $users = loadUsers();
    return isset($users[$username]) && password_verify($password, $users[$username]);
}
