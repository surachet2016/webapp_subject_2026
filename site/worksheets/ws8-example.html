-- ฐานข้อมูลร้านไร่ภูคำ (ใบงานที่ 8)

CREATE DATABASE IF NOT EXISTS phukham_shop
  DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE phukham_shop;

-- หมวดสินค้า
CREATE TABLE categories (
  category_id INT UNSIGNED NOT NULL AUTO_INCREMENT,
  name        VARCHAR(50)  NOT NULL,
  PRIMARY KEY (category_id),
  UNIQUE KEY uq_categories_name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- สินค้า: category_id อ้างอิง categories
CREATE TABLE products (
  product_id  INT UNSIGNED  NOT NULL AUTO_INCREMENT,
  category_id INT UNSIGNED  NOT NULL,
  name        VARCHAR(100)  NOT NULL,
  price       DECIMAL(10,2) NOT NULL,
  stock       INT UNSIGNED  NOT NULL DEFAULT 0,
  image       VARCHAR(255)  NULL,                 -- เก็บเฉพาะชื่อไฟล์ ไฟล์จริงอยู่ใน uploads/products
  created_at  TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (product_id),
  KEY fk_products_category (category_id),
  CONSTRAINT fk_products_category FOREIGN KEY (category_id)
    REFERENCES categories (category_id)
    ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- สมาชิก
CREATE TABLE members (
  member_id     INT UNSIGNED NOT NULL AUTO_INCREMENT,
  username      VARCHAR(30)  NOT NULL,
  password_hash VARCHAR(255) NOT NULL,              -- ค่าจาก password_hash() (ใบงานที่ 7)
  full_name     VARCHAR(100) NOT NULL,
  email         VARCHAR(100) NOT NULL,
  created_at    TIMESTAMP    NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (member_id),
  UNIQUE KEY uq_members_username (username),
  UNIQUE KEY uq_members_email (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- คำสั่งซื้อ: member_id อ้างอิง members
CREATE TABLE orders (
  order_id   INT UNSIGNED  NOT NULL AUTO_INCREMENT,
  member_id  INT UNSIGNED  NOT NULL,
  total      DECIMAL(10,2) NOT NULL,
  created_at TIMESTAMP     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (order_id),
  KEY fk_orders_member (member_id),
  CONSTRAINT fk_orders_member FOREIGN KEY (member_id)
    REFERENCES members (member_id)
    ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

คอลัมน์ชนิดข้อมูลเหตุผล
รหัส (…_id)INT UNSIGNED AUTO_INCREMENTไม่ติดลบ และ MySQL รันเลขให้เอง คอลัมน์ FK ต้องเป็นชนิดเดียวกับ PK ที่อ้างถึง (INT UNSIGNED เหมือนกัน)
price, totalDECIMAL(10,2)เก็บเงินได้ตรงทุกสตางค์ FLOAT และ DOUBLE เก็บทศนิยมแบบประมาณ เช่น 0.1 + 0.2 อาจไม่เท่ากับ 0.3 พอดี
stockINT UNSIGNED DEFAULT 0จำนวนเต็ม ไม่ติดลบ
imageVARCHAR(255) NULLเก็บแค่ชื่อไฟล์ ตัวไฟล์อยู่ในโฟลเดอร์ NULL หมายถึงยังไม่มีรูป
password_hashVARCHAR(255)ความยาวที่ PHP แนะนำสำหรับค่าจาก password_hash() เผื่ออัลกอริทึมในอนาคต
username, emailUNIQUE KEYกันสมัครชื่อหรืออีเมลซ้ำที่ระดับฐานข้อมูล
ทุกตารางENGINE=InnoDB utf8mb4InnoDB จำเป็นสำหรับ Foreign Key และ utf8mb4 เก็บภาษาไทยและอีโมจิได้ครบ

// เชื่อมต่อฐานข้อมูล ใช้ร่วมกันทุกหน้าด้วย require_once
// XAMPP (Windows): รหัสผ่าน root ว่าง ""   |   MAMP (macOS): รหัสผ่าน root คือ "root" (ภาคผนวก ค)
$dbHost = 'localhost';
$dbUser = 'root';
$dbPass = '';
$dbName = 'phukham_shop';

// PHP 8.1 ขึ้นไป: ถ้าคำสั่ง SQL ผิดพลาด mysqli จะโยน mysqli_sql_exception ให้ดักด้วย try/catch ได้
mysqli_report(MYSQLI_REPORT_ERROR | MYSQLI_REPORT_STRICT);

try {
    $conn = mysqli_connect($dbHost, $dbUser, $dbPass, $dbName);
    mysqli_set_charset($conn, 'utf8mb4');      // ภาษาไทยไม่เพี้ยน
} catch (mysqli_sql_exception $e) {
    http_response_code(500);
    // ไม่แสดงรายละเอียด error ให้ผู้ใช้เห็น เพราะอาจเปิดเผยชื่อผู้ใช้หรือโครงสร้างฐานข้อมูล
    exit('เชื่อมต่อฐานข้อมูลไม่ได้ กรุณาตรวจว่าเปิด MySQL แล้ว และค่าใน includes/db.php ถูกต้อง');
}

declare(strict_types=1);

$perPage = 5;
$q = trim($_GET['q'] ?? '');
$like = '%' . $q . '%';                      // เครื่องหมาย % ใส่ในค่า ไม่ใช่ในคำสั่ง SQL

// นับจำนวนทั้งหมดที่ตรงกับคำค้น เพื่อคำนวณจำนวนหน้า
$stmt = mysqli_prepare($conn, 'SELECT COUNT(*) AS total FROM products WHERE name LIKE ?');
mysqli_stmt_bind_param($stmt, 's', $like);
mysqli_stmt_execute($stmt);
$total = (int) mysqli_fetch_assoc(mysqli_stmt_get_result($stmt))['total'];
mysqli_stmt_close($stmt);

$totalPages = max(1, (int) ceil($total / $perPage));
$page = max(1, min($totalPages, (int) ($_GET['page'] ?? 1)));   // บังคับให้อยู่ระหว่าง 1 ถึงหน้าสุดท้าย
$offset = ($page - 1) * $perPage;

$sql = 'SELECT p.product_id, p.name, p.price, p.stock, p.image, c.name AS category
        FROM products p
        JOIN categories c ON c.category_id = p.category_id
        WHERE p.name LIKE ?
        ORDER BY p.product_id
        LIMIT ? OFFSET ?';
$stmt = mysqli_prepare($conn, $sql);
mysqli_stmt_bind_param($stmt, 'sii', $like, $perPage, $offset);   // s = string, i = integer
mysqli_stmt_execute($stmt);
$products = mysqli_fetch_all(mysqli_stmt_get_result($stmt), MYSQLI_ASSOC);
mysqli_stmt_close($stmt);

<?php
// 8.4 ตรวจและบันทึกรูปสินค้า คืนชื่อไฟล์ใหม่ หรือ throw RuntimeException พร้อมข้อความภาษาไทย
declare(strict_types=1);

const UPLOAD_DIR = __DIR__ . '/../uploads/products/';
const MAX_SIZE = 2 * 1024 * 1024;   // 2 MB
const ALLOWED_TYPES = ['image/jpeg' => 'jpg', 'image/png' => 'png', 'image/webp' => 'webp'];

function saveProductImage(array $file): string
{
    if ($file['error'] !== UPLOAD_ERR_OK) {
        throw new RuntimeException($file['error'] === UPLOAD_ERR_INI_SIZE || $file['error'] === UPLOAD_ERR_FORM_SIZE
            ? 'ไฟล์ใหญ่เกินกำหนด' : 'อัปโหลดไม่สำเร็จ (รหัส ' . $file['error'] . ')');
    }
    if ($file['size'] > MAX_SIZE) {
        throw new RuntimeException('ไฟล์ใหญ่เกิน 2 MB');
    }
    // ตรวจชนิดไฟล์จากเนื้อไฟล์จริง ไม่เชื่อนามสกุลหรือชนิดที่เบราว์เซอร์ส่งมา
    $mime = (new finfo(FILEINFO_MIME_TYPE))->file($file['tmp_name']);
    if (!isset(ALLOWED_TYPES[$mime])) {
        throw new RuntimeException('อนุญาตเฉพาะรูป JPG, PNG หรือ WebP');
    }
    // ตั้งชื่อใหม่แบบสุ่ม กันชื่อซ้ำและกันชื่อไฟล์แปลก ๆ จากผู้ใช้
    $newName = bin2hex(random_bytes(16)) . '.' . ALLOWED_TYPES[$mime];
    if (!move_uploaded_file($file['tmp_name'], UPLOAD_DIR . $newName)) {
        throw new RuntimeException('บันทึกไฟล์ไม่สำเร็จ ตรวจสิทธิ์การเขียนโฟลเดอร์ uploads/products');
    }
    return $newName;
}

function deleteProductImage(?string $name): void
{
    // basename กันชื่อไฟล์ที่มี ../ เพื่อไม่ให้ลบไฟล์นอกโฟลเดอร์
    if ($name && is_file(UPLOAD_DIR . basename($name))) {
        unlink(UPLOAD_DIR . basename($name));
    }
}
