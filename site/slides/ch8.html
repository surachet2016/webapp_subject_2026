8.12.1 การทำงานและไวยากรณ์ของ Prepared Statements ใน MySQLi

ส่งโครงคำสั่งก่อน ส่งข้อมูลทีหลัง

1. prepare
SQL มีเครื่องหมาย ?
→
2. bind_param
ผูกค่าพร้อมชนิด
→
3. execute
รันคำสั่ง
→
4. get_result
ดึงผลลัพธ์
$stmt = $conn->prepare( "SELECT CustomerID, Name FROM customer WHERE Name LIKE ? AND Budget >= ?"); $like = "%" . $_GET['q'] . "%"; $min = (float) $_GET['min']; $stmt->bind_param("sd", $like, $min); $stmt->execute(); $result = $stmt->get_result();

ตัวอักษรชนิดใน bind_param

  • s string ข้อความ
  • i integer จำนวนเต็ม
  • d double ทศนิยม
  • b blob ข้อมูลไบนารี

8.14 MySQLi เชิงวัตถุ (OOP) และตัวอย่างโค้ดมาตรฐาน

แบบฟังก์ชัน หรือแบบวัตถุ ผลเหมือนกัน

แบบฟังก์ชัน (Procedural)

$conn = mysqli_connect("localhost", "root", "", "db_test"); mysqli_set_charset($conn, "utf8mb4"); $stmt = mysqli_prepare($conn, "SELECT Name FROM customer WHERE CustomerID = ?"); mysqli_stmt_bind_param($stmt, "s", $id); mysqli_stmt_execute($stmt); $res = mysqli_stmt_get_result($stmt); $row = mysqli_fetch_assoc($res);

แบบวัตถุ (OOP) แนะนำ

$db = new mysqli("localhost", "root", "", "db_test"); $db->set_charset("utf8mb4"); $stmt = $db->prepare( "SELECT Name FROM customer WHERE CustomerID = ?"); $stmt->bind_param("s", $id); $stmt->execute(); $row = $stmt->get_result()->fetch_assoc();

แบบวัตถุสั้นกว่า อ่านเป็นลำดับ และต่อยอดสู่ PDO หรือเฟรมเวิร์กได้ง่าย · ตั้ง mysqli_report(MYSQLI_REPORT_ERROR | MYSQLI_REPORT_STRICT) ให้ข้อผิดพลาดกลายเป็น Exception แล้วจับด้วย try/catch
